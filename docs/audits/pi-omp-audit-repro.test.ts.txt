describe('AUDIT reproduced defects (assertions describe current bugs)', () => {
  it('adopts a fresh external process as an existing tab clear', () => {
    start();
    watcher.registerPending('ui-original', '/proj');
    const original = transcript(0, 'original');
    mockSessionsTree({ project: [original] });
    mockFilesByHeader({ [original]: header('original', '/proj') });
    vi.advanceTimersByTime(2_000);
    expect(mockWriteCliSessionId).toHaveBeenLastCalledWith('ui-original', 'original');
    vi.advanceTimersByTime(10_000);
    const foreign = transcript(0, 'external');
    mockSessionsTree({ project: [original, foreign] });
    mockFilesByHeader({ [original]: header('original', '/proj'), [foreign]: header('external', '/proj') });
    vi.advanceTimersByTime(2_000);
    expect(mockWriteCliSessionId).toHaveBeenLastCalledWith('ui-original', 'external');
  });

  it('lets a resumed registration steal a new tab transcript', () => {
    const old = transcript(10_000, 'resumed');
    mockSessionsTree({ project: [old] });
    start();
    watcher.registerPending('ui-resumed', '/proj');
    watcher.registerPending('ui-new', '/proj');
    const fresh = transcript(0, 'new-session');
    mockSessionsTree({ project: [old, fresh] });
    mockFilesByHeader({ [old]: header('resumed', '/proj'), [fresh]: header('new-session', '/proj') });
    vi.advanceTimersByTime(2_000);
    expect(mockWriteCliSessionId).toHaveBeenCalledWith('ui-resumed', 'new-session');
    expect(mockWriteCliSessionId).not.toHaveBeenCalledWith('ui-new', 'new-session');
  });

  it('reports terminal aborted and length results as working', () => {
    for (const stopReason of ['aborted', 'length']) {
      expect(transcriptStatusFromTail(JSON.stringify({ type: 'message', message: { role: 'assistant', stopReason } }))).toBe('working');
    }
  });

  it('takes one entire history snapshot per registered UI session', () => {
    const old = Array.from({ length: 1000 }, (_, i) => `old-${i}.jsonl`);
    mockSessionsTree({ project: old });
    start();
    const before = stats.rootReads;
    for (let i = 0; i < 10; i++) watcher.registerPending(`ui-${i}`, '/proj');
    expect(stats.rootReads - before).toBe(10);
  });
});

const mockReaddirSync = vi.mocked(fs.readdirSync);
const mockOpenSync = vi.mocked(fs.openSync);
const mockReadSync = vi.mocked(fs.readSync);
const mockCloseSync = vi.mocked(fs.closeSync);
const mockWriteCliSessionId = vi.mocked(writeCliSessionId);
const mockWatch = vi.mocked(fs.watch);

const ROOT = '/mock/sessions';

/**
 * Work counters. `rootReads` counts one full-tree walk (a scan re-reads the
 * root first, so it is a direct proxy for "how many scans ran"); `opened`
 * records every transcript the scan actually opened, which is what the
 * filename-timestamp floor is supposed to shrink.
 */
const stats = { rootReads: 0, opened: [] as string[] };

/**
 * Mock the sessions tree: readdirSync(ROOT) returns the dir names,
 * readdirSync(ROOT/<dir>) returns that dir's files. Anything else throws
 * ENOENT, as a real missing path would.
 */
function mockSessionsTree(filesByDir: Record<string, string[]>): void {
  mockReaddirSync.mockImplementation((p: string) => {
    if (p === ROOT) {
      stats.rootReads += 1;
      return Object.keys(filesByDir);
    }
    for (const [dir, files] of Object.entries(filesByDir)) {
      if (p === path.join(ROOT, dir)) return files;
    }
    throw new Error('ENOENT');
  });
}

/** Serve a different header line per basename, so coexisting files keep their own ids. */
function mockFilesByHeader(lineByBasename: Record<string, string>): void {
  let fdCounter = 100;
  const fdToLine = new Map<number, string>();
  mockOpenSync.mockImplementation((p: string) => {
    stats.opened.push(p);
    const line = lineByBasename[path.basename(p)];
    if (line === undefined) return -1;
    const fd = fdCounter++;
    fdToLine.set(fd, line);
    return fd;
  });
  mockReadSync.mockImplementation((fd: number, target: Buffer) => {
    const line = fdToLine.get(fd);
    if (line === undefined) return 0;
    const buf = Buffer.from(line);
    buf.copy(target);
    return buf.length;
  });
  mockCloseSync.mockReturnValue(true);
}

function header(id: string, cwd: string): string {
  return JSON.stringify({ type: 'session', version: 3, id, timestamp: 't', cwd });
}

/** `<ISO timestamp>_<id>.jsonl` stamped `ageMs` before the (fake) current time. */
function transcript(ageMs: number, id: string): string {
  const d = new Date(Date.now() - ageMs);
  const p = (n: number) => String(n).padStart(2, '0');
  const ms = String(d.getUTCMilliseconds()).padStart(3, '0');
  return `${d.getUTCFullYear()}-${p(d.getUTCMonth() + 1)}-${p(d.getUTCDate())}T${p(d.getUTCHours())}-${p(d.getUTCMinutes())}-${p(d.getUTCSeconds())}-${ms}Z_${id}.jsonl`;
}

const MINUTE = 60_000;

let watcher: CompatibleSessionWatcher;

function start(): void {
  mockWatch.mockReturnValue({ close: vi.fn() } as unknown as fs.FSWatcher);
  watcher = createCompatibleSessionWatcher(() => ROOT, { onAdopted });
  watcher.start();
}

/** The single callback every watcher (root + subdirs) is wired to. */
function watchEvent(): (...args: unknown[]) => void {
  return mockWatch.mock.calls[0][1] as (...args: unknown[]) => void;
}

beforeEach(() => {
  vi.clearAllMocks();
  vi.useFakeTimers();
  stats.rootReads = 0;
  stats.opened = [];
  electronMock.windows = [{ isDestroyed: () => false }];
  mockSessionsTree({});
});

afterEach(() => {
  watcher?.stop();
  vi.useRealTimers();
});

describe('event-driven scan coalescing', () => {
  it('collapses a burst of watch events into a single scan', () => {
    mockSessionsTree({ 'dir-a': [] });
    start();
    mockSessionsTree({ 'dir-a': [] });
    watcher.registerPending('ui-1', '/proj');

    const before = stats.rootReads;
    for (let i = 0; i < 50; i += 1) watchEvent()();
    expect(stats.rootReads).toBe(before); // nothing ran synchronously

    vi.advanceTimersByTime(250);
    expect(stats.rootReads - before).toBe(1);
  });

  it('bounds the scan rate under a continuous stream of events', () => {
    mockSessionsTree({ 'dir-a': [] });
    start();
    mockSessionsTree({ 'dir-a': [] });
    watcher.registerPending('ui-1', '/proj');

    const before = stats.rootReads;
    // 5s of events every 10ms — hundreds of raw events.
    for (let i = 0; i < 500; i += 1) {
      watchEvent()();
      vi.advanceTimersByTime(10);
    }
    const scans = stats.rootReads - before;

    // Never starved (at least one scan per second of the burst) and never
    // proportional to the event count (500 raw events).
    expect(scans).toBeGreaterThanOrEqual(5);
    expect(scans).toBeLessThanOrEqual(10);
  });

  it('still discovers a new transcript on the coalesced event path', () => {
    mockSessionsTree({ 'dir-a': [] });
    start();
    const fresh = transcript(1_000, 'cli-fast');
    watcher.registerPending('ui-1', '/proj');
    mockSessionsTree({ 'dir-a': [fresh] });
    mockFilesByHeader({ [fresh]: header('cli-fast', '/proj') });

    watchEvent()();
    vi.advanceTimersByTime(250);

    expect(mockWriteCliSessionId).toHaveBeenCalledWith('ui-1', 'cli-fast');
  });

  it('does not scan at all when no session is registered', () => {
    mockSessionsTree({ 'dir-a': ['x.jsonl'] });
    start();

    const before = stats.rootReads;
    watchEvent()();
    vi.advanceTimersByTime(5_000);

    expect(stats.rootReads).toBe(before);
    expect(mockOpenSync).not.toHaveBeenCalled();
  });

  it('drops a pending coalesced scan on stop()', () => {
    mockSessionsTree({ 'dir-a': [] });
    start();
    watcher.registerPending('ui-1', '/proj');
    mockSessionsTree({ 'dir-a': [transcript(1_000, 'cli-late')] });
    mockFilesByHeader({});

    const before = stats.rootReads;
    watchEvent()();
    watcher.stop();
    vi.advanceTimersByTime(10_000);

    expect(stats.rootReads).toBe(before);
  });
});

describe('scan cost is bounded by recency', () => {
  it('never opens transcripts older than every session lower bound', () => {
    const ancient = transcript(30 * MINUTE, 'cli-ancient');
    const stale = transcript(10 * MINUTE, 'cli-stale');
    const fresh = transcript(1_000, 'cli-fresh');
    mockSessionsTree({ 'dir-a': [] });
    start();
    watcher.registerPending('ui-1', '/proj');
    // All three transcripts appear AFTER registration, so the
    // registration-time `knownFiles` snapshot excludes none of them. The
    // only thing that can keep the old ones off disk is the filename floor.
    mockSessionsTree({ 'dir-a': [ancient, stale, fresh] });
    mockFilesByHeader({
      [ancient]: header('cli-ancient', '/proj'),
      [stale]: header('cli-stale', '/proj'),
      [fresh]: header('cli-fresh', '/proj'),
    });

    vi.advanceTimersByTime(2_000);

    // Exactly one open, and it is the recent one: the old transcripts are
    // rejected from the filename alone — no open, no 8KB read, no parse.
    // (Names are `<ISO timestamp>_<id>.jsonl`, so compare whole basenames;
    // `path.basename` never yields a bare `cli-fresh`.)
    expect(stats.opened.map((p) => path.basename(p))).toEqual([fresh]);
    expect(mockWriteCliSessionId).toHaveBeenCalledWith('ui-1', 'cli-fresh');
  });

  it('still opens files whose name carries no timestamp', () => {
    const legacy = 'no-timestamp.jsonl';
    mockSessionsTree({ 'dir-a': [] });
    start();
    watcher.registerPending('ui-1', '/proj');
    mockSessionsTree({ 'dir-a': [legacy] });
    mockFilesByHeader({ [legacy]: header('cli-legacy', '/proj') });

    vi.advanceTimersByTime(2_000);

    expect(stats.opened).toContain(path.join(ROOT, 'dir-a', legacy));
    expect(mockWriteCliSessionId).toHaveBeenCalledWith('ui-1', 'cli-legacy');
  });
});

describe('/clear re-adoption freshness bound', () => {
  /**
   * Register `ui-1` against an empty tree, let it adopt `first`, then idle
   * `ms` with nothing else happening. The transcript has to appear AFTER
   * `registerPending`: anything already on disk at registration is snapshotted
   * into `knownFiles` and is never adoptable by design.
   */
  function withAdoptedSession(first: string, ms: number): void {
    mockSessionsTree({ 'dir-a': [] });
    start();
    watcher.registerPending('ui-1', '/proj');
    mockSessionsTree({ 'dir-a': [first] });
    mockFilesByHeader({ [first]: header('cli-first', '/proj') });
    vi.advanceTimersByTime(2_000);
    expect(mockWriteCliSessionId).toHaveBeenLastCalledWith('ui-1', 'cli-first');
    vi.advanceTimersByTime(ms);
    mockWriteCliSessionId.mockClear();
  }

  it('rejects an unrelated external run in the same cwd', () => {
    // The session adopted shortly after registration, then just sat there
    // for eleven minutes.
    const first = transcript(1_000, 'cli-first');
    withAdoptedSession(first, 11 * MINUTE);

    // Someone launches `omp`/`pi` directly in the same project directory.
    // It is newer than the adopted transcript and absent from the
    // registration-time knownFiles snapshot — but it is not a `/clear`.
    const external = transcript(5 * MINUTE, 'cli-external');
    mockSessionsTree({ 'dir-a': [first, external] });
    mockFilesByHeader({
      [first]: header('cli-first', '/proj'),
      [external]: header('cli-external', '/proj'),
    });

    vi.advanceTimersByTime(10_000);

    expect(mockWriteCliSessionId).not.toHaveBeenCalled();
    expect(onAdopted).not.toHaveBeenCalledWith('ui-1', 'cli-external', '/proj', undefined);
  });

  it('the clear window alone rejects a stale external run that survives the scan floor', () => {
    // Two sessions register against an empty tree: ui-1 in /proj, ui-2 in
    // /other. Only ui-1 ever gets a transcript, so ui-2 stays pending.
    mockSessionsTree({ 'dir-a': [] });
    start();
    watcher.registerPending('ui-1', '/proj');
    watcher.registerPending('ui-2', '/other');
    const first = transcript(1_000, 'cli-first');
    mockSessionsTree({ 'dir-a': [first] });
    mockFilesByHeader({ [first]: header('cli-first', '/proj') });
    vi.advanceTimersByTime(2_000);
    expect(mockWriteCliSessionId).toHaveBeenLastCalledWith('ui-1', 'cli-first');
    mockWriteCliSessionId.mockClear();

    // Ten minutes on, an unrelated CLI run appears in /proj. Because the
    // still-pending ui-2 drags the scan floor back to its own (old)
    // registration bound, this file is NOT filtered by filename — it gets
    // opened and becomes a real candidate. Only CLEAR_ADOPTION_WINDOW_MS
    // can stop it hijacking ui-1's tab.
    vi.advanceTimersByTime(10 * MINUTE);
    const external = transcript(5 * MINUTE, 'cli-external');
    mockSessionsTree({ 'dir-a': [first, external] });
    mockFilesByHeader({
      [first]: header('cli-first', '/proj'),
      [external]: header('cli-external', '/proj'),
    });

    vi.advanceTimersByTime(10_000);

    // The scan floor now rejects old external transcripts before opening them.
    expect(stats.opened.map((p) => path.basename(p))).not.toContain(external);
    expect(mockWriteCliSessionId).not.toHaveBeenCalled();
    expect(onAdopted).not.toHaveBeenCalledWith('ui-1', 'cli-external', '/proj', undefined);
  });

  it('adopts a genuine /clear transcript promptly', () => {
    const first = transcript(1_000, 'cli-first');
    withAdoptedSession(first, 11 * MINUTE);

    const cleared = transcript(2_000, 'cli-cleared');
    mockSessionsTree({ 'dir-a': [first, cleared] });
    mockFilesByHeader({
      [first]: header('cli-first', '/proj'),
      [cleared]: header('cli-cleared', '/proj'),
    });

    watchEvent()();
    // A full SCAN_MIN_SPACING_MS: the 2s poll had just scanned when the
    // idle period ended, so the coalesced event-driven scan cannot start
    // sooner than the minimum spacing allows (debounce 250 < spacing 1000).
    vi.advanceTimersByTime(1_000);

    expect(mockWriteCliSessionId).toHaveBeenCalledWith('ui-1', 'cli-cleared');
    expect(onAdopted).toHaveBeenCalledWith('ui-1', 'cli-cleared', '/proj', undefined);
  });

  it('re-adopts a clear created in the same second as the original transcript', () => {
    vi.setSystemTime(new Date('2026-09-29T00:00:00.100Z'));
    mockSessionsTree({ 'dir-a': [] });
    start();
    watcher.registerPending('ui-1', '/proj');
    const first = transcript(0, 'cli-first');
    mockSessionsTree({ 'dir-a': [first] });
    mockFilesByHeader({ [first]: header('cli-first', '/proj') });
    watchEvent()();
    vi.advanceTimersByTime(250);
    expect(mockWriteCliSessionId).toHaveBeenCalledWith('ui-1', 'cli-first');
    mockWriteCliSessionId.mockClear();

    const cleared = transcript(0, 'cli-cleared');
    mockSessionsTree({ 'dir-a': [first, cleared] });
    mockFilesByHeader({
      [first]: header('cli-first', '/proj'),
      [cleared]: header('cli-cleared', '/proj'),
    });
    watchEvent()();
    vi.advanceTimersByTime(1_000);
    expect(mockWriteCliSessionId).toHaveBeenCalledWith('ui-1', 'cli-cleared');
  });

  it('keeps the ambiguity rule: no re-adoption with two adopted sessions in the cwd', () => {
    const a = transcript(1_000, 'cli-a');
    const b = transcript(1_000, 'cli-b');
    mockSessionsTree({ 'dir-a': [] });
    start();
    watcher.registerPending('ui-A', '/proj');
    watcher.registerPending('ui-B', '/proj');
    mockSessionsTree({ 'dir-a': [a, b] });
    mockFilesByHeader({ [a]: header('cli-a', '/proj'), [b]: header('cli-b', '/proj') });
    vi.advanceTimersByTime(2_000);
    expect(mockWriteCliSessionId).toHaveBeenCalledWith('ui-A', 'cli-a');
    expect(mockWriteCliSessionId).toHaveBeenCalledWith('ui-B', 'cli-b');
    mockWriteCliSessionId.mockClear();

    const cleared = transcript(2_000, 'cli-clear');
    mockSessionsTree({ 'dir-a': [a, b, cleared] });
    mockFilesByHeader({
      [a]: header('cli-a', '/proj'),
      [b]: header('cli-b', '/proj'),
      [cleared]: header('cli-clear', '/proj'),
    });

    vi.advanceTimersByTime(2_000);

    expect(mockWriteCliSessionId).not.toHaveBeenCalled();
  });
});

describe('Pass 1 / Pass 2 candidate collision', () => {
  it('never hands a transcript already claimed by a pending session to another session', () => {
    // A registers against an empty root, then its transcript appears and is adopted.
    mockSessionsTree({ 'dir-a': [] });
    start();
    watcher.registerPending('ui-A', '/proj');
    const a = transcript(3_000, 'cli-a');
    mockSessionsTree({ 'dir-a': [a] });
    mockFilesByHeader({ [a]: header('cli-a', '/proj') });
    vi.advanceTimersByTime(2_000);
    expect(mockWriteCliSessionId).toHaveBeenCalledWith('ui-A', 'cli-a');
    mockWriteCliSessionId.mockClear();

    // B registers afterwards (so `a` is already in its knownFiles), then B's
    // own transcript appears.
    watcher.registerPending('ui-B', '/proj');
    const b = transcript(1_000, 'cli-b');
    mockSessionsTree({ 'dir-a': [a, b] });
    mockFilesByHeader({
      [a]: header('cli-a', '/proj'),
      [b]: header('cli-b', '/proj'),
    });

    vi.advanceTimersByTime(2_000);

    // Pass 1 correctly gives B its own transcript. Pass 2 must then NOT drag
    // A onto that same file: two sessions sharing one cliSessionId resolve to
    // the same transcript, so every title that transcript publishes renames
    // both tabs on every surface — the reported "auto-naming renames all
    // sessions of the current project".
    expect(mockWriteCliSessionId).toHaveBeenCalledWith('ui-B', 'cli-b');
    const written = mockWriteCliSessionId.mock.calls.map(
      ([ui, cli]) => `${ui}=${cli}`
    );
    expect(written).not.toContain('ui-A=cli-b');
  });
});
